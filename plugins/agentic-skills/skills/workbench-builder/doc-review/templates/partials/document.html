{#- The memo, rendered block-by-block. Each block carries data-block="" and renders ONLY its normalized plain text (b.text) — no stray whitespace. The browser's textContent of the block is therefore byte-identical to b.text, so selection offsets index the same string the server stored, and a painted lands on exactly the selected characters. CRITICAL: every Jinja tag here uses whitespace trimming ({%- ... -%}, {{- ... -}}) so the macro injects no newlines/indentation between text runs. A single stray newline shifts every downstream offset. -#} {%- macro render_block(b, anns) -%} {%- set text = b.text -%} {%- if anns -%} {%- set ns = namespace(cursor=0) -%} {%- for a in anns|sort(attribute='start_off') -%} {%- if a.start_off >= ns.cursor and a.end_off <= text|length and a.end_off > a.start_off -%} {%- set cls = ('mk-redline' if a.kind == 'redline' else 'mk-comment') + (' mk-done' if a.status != 'open' else '') -%} {{- text[ns.cursor:a.start_off] -}} {{- text[a.start_off:a.end_off] -}} {%- set ns.cursor = a.end_off -%} {%- endif -%} {%- endfor -%} {{- text[ns.cursor:] -}} {%- else -%} {{- text -}} {%- endif -%} {%- endmacro -%}
{%- for b in blocks %} <{{ b.tag }} class="blk blk-{{ b.tag }}" data-block="{{ b.id }}">{{- render_block(b, by_block.get(b.id, [])) -}} {%- endfor %}
